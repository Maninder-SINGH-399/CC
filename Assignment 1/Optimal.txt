#include <iostream>
using namespace std;

int main() {
    int n, target;
    cin >> n;

    int a[100];
    for (int i = 0; i < n; i++)
        cin >> a[i];

    cin >> target;

    int left = 0, right = n - 1;
    bool found = false;

    while (left <= right) {
        int mid = (left + right) / 2;

        if (a[mid] == target) {
            found = true;
            break;
        }

        if (a[left] == a[mid] && a[mid] == a[right]) {
            left++;
            right--;
        }
        else if (a[left] <= a[mid]) {
            if (a[left] <= target && target < a[mid])
                right = mid - 1;
            else
                left = mid + 1;
        }
        else {
            if (a[mid] < target && target <= a[right])
                left = mid + 1;
            else
                right = mid - 1;
        }
    }

    if (found)
        cout << "True";
    else
        cout << "False";

    return 0;
}
