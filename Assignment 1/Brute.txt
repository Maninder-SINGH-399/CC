#include <iostream>
using namespace std;

int main() {
    int n, target;
    cin >> n;

    int a[100];
    for (int i = 0; i < n; i++)
        cin >> a[i];

    cin >> target;

    bool found = false;

    for (int i = 0; i < n; i++) {
        if (a[i] == target) {
            found = true;
            break;
        }
    }

    if (found)
        cout << "True";
    else
        cout << "False";

    return 0;
}
