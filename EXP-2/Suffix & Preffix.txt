#include <iostream>
#include <vector>
using namespace std;

int main() {
    int a;
    cin >> a;

    vector<int> nums(a), ans(a,1);

    for (int i = 0; i < a; i++)
        cin >> nums[i];

    for (int i = 1; i < a; i++)
        ans[i] = ans[i-1] * nums[i-1];

    int right = 1;
    for (int i = a-1; i>= 0; i--) {
        ans[i] *= right;
        right *= nums[i];
    }

    for (int x : ans)
        cout << x << " ";

    return 0;
}
