#include <iostream>
#include <vector>
using namespace std;

int findPosition(vector<int> arr, int key) {
    int start = 0, end = arr.size() - 1;

    while (start <= end) {
        int middle = (start + end) / 2;

        if (arr[middle] == key)
            return middle;
        else if (arr[middle] < key)
            start = middle + 1;
        else
            end = middle - 1;
    }

    return start;
}

int main() {
    int n, key;

    cout << "Enter size of array: ";
    cin >> n;

    vector<int> arr(n);

    cout << "Enter sorted array elements: ";
    for (int i = 0; i < n; i++)
        cin >> arr[i];

    cout << "Enter target element: ";
    cin >> key;

    cout << "Insert Position: " << findPosition(arr, key);

    return 0;
}
