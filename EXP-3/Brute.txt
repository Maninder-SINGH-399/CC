#include <iostream>
using namespace std;

int main() {
    int n, x;

    cout << "Enter size: ";
    cin >> n;

    int a[n];

    cout << "Enter sorted elements: ";
    for (int i = 0; i < n; i++)
        cin >> a[i];

    cout << "Enter target: ";
    cin >> x;

    int pos = n;

    for (int i = 0; i < n; i++) {
        if (a[i] >= x) {
            pos = i;
            break;
        }
    }

    cout << "Insert Position: " << pos;

    return 0;
}
