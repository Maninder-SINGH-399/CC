#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
};

class Solution {
public:
    ListNode* detectCycle(ListNode* head) {
        ListNode *slow = head, *fast = head;

        while (fast && fast->next) {
            slow = slow->next;
            fast = fast->next->next;

            if (slow == fast) {
                slow = head;

                while (slow != fast) {
                    slow = slow->next;
                    fast = fast->next;
                }

                return slow;
            }
        }

        return NULL;
    }
};

int main() {
    ListNode a{1}, b{2}, c{3}, d{4};

    a.next = &b;
    b.next = &c;
    c.next = &d;
    d.next = &b;

    Solution s;
    ListNode* result = s.detectCycle(&a);

    if (result)
        cout << "Cycle starts at node: " << result->val;
    else
        cout << "No cycle";

    return 0;
}
