#include <bits/stdc++.h>
using namespace std;

int main() {
    vector<int> heights = {2, 1, 5, 6, 2, 3};
    stack<int> st;
    int n = heights.size();
    int maxi = 0;

    for (int i = 0; i <= n; i++) {
        int curr = (i == n) ? 0 : heights[i];

        while (!st.empty() && heights[st.top()] > curr) {
            int h = heights[st.top()];
            st.pop();

            int width;
            if (st.empty())
                width = i;
            else
                width = i - st.top() - 1;

            maxi = max(maxi, h * width);
        }

        if (i < n)
            st.push(i);
    }

    cout << "Largest Rectangle Area: " << maxi << endl;

    return 0;
}
