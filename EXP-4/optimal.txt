#include <iostream>
using namespace std;

int main() {
    int n, k, target;
    cin >> n;

    int a[100];
    for (int i = 0; i < n; i++)
        cin >> a[i];

    cin >> k >> target;
    k = k % n;

    for (int i = 0; i < k; i++) {
        int first = a[0];
        for (int j = 0; j < n - 1; j++)
            a[j] = a[j + 1];
        a[n - 1] = first;
    }

    cout << "Rotated Array: ";
    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    int l = 0, r = n - 1, ans = -1;

    while (l <= r) {
        int m = (l + r) / 2;

        if (a[m] == target) {
            ans = m;
            break;
        }

        if (a[l] <= a[m]) {
            if (a[l] <= target && target < a[m])
                r = m - 1;
            else
                l = m + 1;
        }
        else {
            if (a[m] < target && target <= a[r])
                l = m + 1;
            else
                r = m - 1;
        }
    }

    if (ans != -1)
        cout << "\nTarget found at index: " << ans;
    else
        cout << "\nTarget not found.";

    return 0;
}
